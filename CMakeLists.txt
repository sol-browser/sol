cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" RED_APP_VERSION)
string(STRIP "${RED_APP_VERSION}" RED_APP_VERSION)
if(NOT RED_APP_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain major.minor.patch")
endif()
project(SolBrowser VERSION ${RED_APP_VERSION} LANGUAGES C CXX)
foreach(part PROJECT_VERSION_MAJOR PROJECT_VERSION_MINOR PROJECT_VERSION_PATCH)
  if(${part} GREATER 65535)
    message(FATAL_ERROR "VERSION components must fit Windows' 16-bit version fields")
  endif()
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This milestone supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CEF_ROOT "${CMAKE_SOURCE_DIR}/third_party/cef" CACHE PATH "Verified CEF distribution")
if(NOT EXISTS "${CEF_ROOT}/.archive-sha256")
  message(FATAL_ERROR "Run python scripts/fetch_cef.py first")
endif()
file(READ "${CMAKE_SOURCE_DIR}/cmake/cef.lock.json" CEF_LOCK)
string(JSON EXPECTED_SHA256 GET "${CEF_LOCK}" sha256)
file(READ "${CEF_ROOT}/.archive-sha256" EXTRACTED_SHA256)
string(STRIP "${EXTRACTED_SHA256}" EXTRACTED_SHA256)
if(NOT EXTRACTED_SHA256 STREQUAL EXPECTED_SHA256)
  message(FATAL_ERROR "CEF extraction does not match lock")
endif()
set(USE_SANDBOX ON CACHE BOOL "Required Chromium sandbox")
if(NOT USE_SANDBOX)
  message(FATAL_ERROR "Sandbox cannot be disabled")
endif()
set(USE_ATL OFF CACHE BOOL "No ATL dependency")
set(PROJECT_ARCH x86_64)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
# Red's engine patches (cef-patches/) add experimental API: build the client
# and the wrapper against the experimental API when the distribution has it.
# Client and libcef come from the same build, so their API hashes match.
file(READ "${CEF_ROOT}/include/cef_browser.h" RED_CEF_BROWSER_H)
if(RED_CEF_BROWSER_H MATCHES "RunExtensionAction")
  list(REMOVE_ITEM CEF_COMPILER_DEFINES "CEF_API_VERSION=CEF_API_VERSION_LAST")
  list(APPEND CEF_COMPILER_DEFINES "CEF_API_VERSION=CEF_EXPERIMENTAL")
  message(STATUS "CEF: Sol-patched distribution, experimental API")
endif()
# Engine patch 12 (translucent Views window, Acrylic): only when the distribution has it.
file(READ "${CEF_ROOT}/include/views/cef_window_delegate.h" RED_CEF_WINDOW_DELEGATE_H)
# Engine patch 11 (live page transform): Link Peek keeps its source pages live.
file(READ "${CEF_ROOT}/include/cef_browser.h" RED_CEF_BROWSER_H)
include(cmake/cef_content_transform.cmake)
red_check_cef_content_transform("${RED_CEF_BROWSER_H}" RED_CEF_HAS_CONTENT_TRANSFORM)
if(RED_CEF_HAS_CONTENT_TRANSFORM)
  add_compile_definitions(RED_CEF_CONTENT_TRANSFORM=1)
  message(STATUS "CEF: live page transform available")
endif()
red_check_cef_page_container("${RED_CEF_BROWSER_H}" RED_CEF_HAS_PAGE_CONTAINER)
if(RED_CEF_HAS_PAGE_CONTAINER)
  add_compile_definitions(RED_CEF_PAGE_CONTAINER=1)
  message(STATUS "CEF: persistent page container available")
endif()
if(RED_CEF_HAS_PAGE_CONTAINER AND RED_CEF_BROWSER_H MATCHES "virtual void SetContentContainerCornerRadius\\(int radius\\)")
  add_compile_definitions(RED_CEF_CONTAINER_CORNERS=1)
  message(STATUS "CEF: persistent resting page clip available")
endif()
if(RED_CEF_WINDOW_DELEGATE_H MATCHES "IsTranslucent")
  add_compile_definitions(RED_CEF_TRANSLUCENT=1)
  message(STATUS "CEF: translucent windows available")
endif()
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
enable_testing()
include(cmake/sqlite.cmake)
include(cmake/velopack.cmake)
add_subdirectory(apps/desktop)
